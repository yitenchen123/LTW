cmake_minimum_required(VERSION 3.16)
project(ltw C CXX)

# -------- Platform detection --------
# LTW is a GL-core -> GLES translator. It dlopens a host libEGL at runtime via
# the LIBGL_EGL env var, so we do NOT link any system GL/EGL here -- only the
# headers are needed, and we ship our own (ltw/src/main/tinywrapper/GL and the
# glsl_optimizer/include tree). This is what makes the same source build on
# Android, iOS, macOS and Linux.

set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()

# Hide all non-API symbols by default; only gl*/egl* are exported via the
# version script (Linux) / exported_symbols list (Apple).
set(CMAKE_C_VISIBILITY_PRESET hidden)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)

# -------- Common compile flags (mirrors ltw/src/main/tinywrapper/Android.mk) --------
set(LTW_COMMON_FLAGS
    -D_LIB
    -DNOMINMAX
    -D_USE_MATH_DEFINES
    -D__STDC_NO_THREADS__
    -D__STDC_LIMIT_MACROS
    -D__STDC_FORMAT_MACROS
    -D__STDC_CONSTANT_MACROS
    -DUTIL_ARCH_LITTLE_ENDIAN
    -DUNIX
    -DHAVE_OPENGL
    -DHAVE_OPENGL_ES_1
    -DHAVE_OPENGL_ES_2
    -fvisibility=hidden
)

# Android-specific defines from the original Android.mk. On non-Android we
# intentionally do NOT define DETECT_OS_ANDROID/HAVE_STRUCT_TIMESPEC/DETECT_OS_ANDROID,
# so mesa's detect_os.h auto-detects Linux/Apple instead and picks the right
# sysctl/proc paths. We only carry over the include-directory shims.
if(ANDROID)
    list(APPEND LTW_COMMON_FLAGS
        -DANDROID
        -DHAVE_STRUCT_TIMESPEC
        -DDETECT_OS_ANDROID
    )
endif()

# -------- Include paths --------
set(LTW_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/ltw/src/main/tinywrapper)
set(GLSL_OPT_ROOT ${LTW_ROOT}/glsl_optimizer)

set(LTW_INCLUDE_DIRS
    ${LTW_ROOT}
    ${GLSL_OPT_ROOT}/include
    ${GLSL_OPT_ROOT}/src
    ${GLSL_OPT_ROOT}/src/compiler
    ${GLSL_OPT_ROOT}/src/compiler/nir
    ${GLSL_OPT_ROOT}/src/compiler/glsl
    ${GLSL_OPT_ROOT}/src/compiler/glsl/glcpp
    ${GLSL_OPT_ROOT}/src/gallium/auxiliary
    ${GLSL_OPT_ROOT}/src/gallium/include
    ${GLSL_OPT_ROOT}/src/mesa
    ${GLSL_OPT_ROOT}/src/util
    ${GLSL_OPT_ROOT}/src/util/format
)

include_directories(${LTW_INCLUDE_DIRS})

# -------- glsl_optimizer static library (mirrors Android.mk LOCAL_SRC_FILES) --------
# Generated by lifting the source list verbatim from ltw/src/main/tinywrapper/Android.mk.
file(GLOB_RECURSE GLSL_OPT_SRCS CONFIGURE_DEPENDS
    ${GLSL_OPT_ROOT}/src/code/*.cpp
    ${GLSL_OPT_ROOT}/src/code/*.c
    ${GLSL_OPT_ROOT}/src/util/*.c
    ${GLSL_OPT_ROOT}/src/util/*.cpp
    ${GLSL_OPT_ROOT}/src/util/format/*.c
    ${GLSL_OPT_ROOT}/src/util/sha1/*.c
    ${GLSL_OPT_ROOT}/src/mesa/main/*.c
    ${GLSL_OPT_ROOT}/src/mesa/program/*.c
    ${GLSL_OPT_ROOT}/src/compiler/shader_enums.c
    ${GLSL_OPT_ROOT}/src/compiler/glsl_types.cpp
    ${GLSL_OPT_ROOT}/src/compiler/nir_types.cpp
    ${GLSL_OPT_ROOT}/src/compiler/glsl/*.cpp
    ${GLSL_OPT_ROOT}/src/compiler/glsl/*.c
    ${GLSL_OPT_ROOT}/src/compiler/glsl/glcpp/*.c
    ${GLSL_OPT_ROOT}/src/compiler/nir/*.c
)

add_library(glsl_optimizer STATIC ${GLSL_OPT_SRCS})
target_compile_definitions(glsl_optimizer PRIVATE ${LTW_COMMON_FLAGS})
target_compile_options(glsl_optimizer PRIVATE
    -ffunction-sections
    -fdata-sections
)

if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    target_compile_options(glsl_optimizer PRIVATE -Wno-error)
endif()

# -------- ltw shared library (mirrors LOCAL_SRC_FILES of `ltw` module) --------
set(LTW_SOURCES
    ${LTW_ROOT}/egl.c
    ${LTW_ROOT}/proc.c
    ${LTW_ROOT}/main.c
    ${LTW_ROOT}/glformats.c
    ${LTW_ROOT}/basevertex.c
    ${LTW_ROOT}/blending.c
    ${LTW_ROOT}/query.c
    ${LTW_ROOT}/shader_wrapper.c
    ${LTW_ROOT}/string_utils.c
    ${LTW_ROOT}/framebuffer.c
    ${LTW_ROOT}/of_buffer_copier.c
    ${LTW_ROOT}/stubs.c
    ${LTW_ROOT}/multidraw.c
    ${LTW_ROOT}/vertexattrib.c
    ${LTW_ROOT}/swizzle.c
    ${LTW_ROOT}/license_notice.c
    ${LTW_ROOT}/env.c
    ${LTW_ROOT}/vgpu_shaderconv/shaderconv.c
    ${LTW_ROOT}/unordered_map/unordered_map.c
    ${LTW_ROOT}/unordered_map/int_hash.c
)

add_library(ltw SHARED ${LTW_SOURCES})
target_compile_definitions(ltw PRIVATE ${LTW_COMMON_FLAGS})
target_compile_options(ltw PRIVATE
    -ffunction-sections
    -fdata-sections
)
target_link_libraries(ltw PRIVATE glsl_optimizer)

# Export only gl*/egl* symbols, matching version.script on ELF. On Mach-O we
# use an exported-symbols list so non-API mesa symbols stay hidden.
if(APPLE)
    target_link_options(ltw PRIVATE
        -Wl,-dead_strip
        LINKER "-exported_symbols_list" "${CMAKE_CURRENT_SOURCE_DIR}/ltw/exported_symbols"
    )
else()
    target_link_options(ltw PRIVATE
        -Wl,--version-script=${LTW_ROOT}/version.script
        -Wl,--gc-sections
    )
    # LTO matches Android.mk "Comment for debugging" line
    target_link_options(ltw PRIVATE -flto)
endif()

# pthread is needed by mesa's c11 threads shim and by thread_local in egl.c
find_package(Threads REQUIRED)
target_link_libraries(ltw PRIVATE Threads::Threads)

# On Android we still want liblog for __android_log_print. Other platforms
# route through stderr via ltw_log.h and need nothing extra.
if(ANDROID)
    target_link_libraries(ltw PRIVATE log)
endif()

set_target_properties(ltw PROPERTIES
    OUTPUT_NAME ltw
    PREFIX lib
)

install(TARGETS ltw LIBRARY DESTINATION lib)
